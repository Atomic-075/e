import java.util.Scanner;
import java.util.Arrays;

public class itemPrices{
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        double[] prices = new double[12];
        int size = 10;

        // a. Enter 10 prices
        for (int i = 0; i < size; i++) {
            System.out.print("Enter price " + (i + 1) + ": ");
            prices[i] = sc.nextDouble();
        }

        // b(i). Insert at beginning
        System.out.print("Enter new price: ");
        double newPrice = sc.nextDouble();

        for (int i = size; i > 0; i--) {
            prices[i] = prices[i - 1];
        }

        prices[0] = newPrice;
        size++;

        // b(ii). Insert at specific index
        System.out.print("Enter price to insert: ");
        double p = sc.nextDouble();

        System.out.print("Enter index: ");
        int index = sc.nextInt();

        if (index >= 0 && index <= size) {
            for (int i = size; i > index; i--) {
                prices[i] = prices[i - 1];
            }

            prices[index] = p;
            size++;
        } else {
            System.out.println("Invalid index!");
        }

        // c(i). Display input order
        System.out.println("Prices:");
        for (int i = 0; i < size; i++) {
            System.out.print(prices[i] + " ");
        }

        // c(ii). Display reverse order
        System.out.println("\nReverse order:");
        for (int i = size - 1; i >= 0; i--) {
            System.out.print(prices[i] + " ");
        }

        // d. Calculate total
        double total = 0;

        for (int i = 0; i < size; i++) {
            total += prices[i];
        }

        System.out.println("\nTotal = " + total);

        // e. Binary search
        Arrays.sort(prices, 0, size);

        System.out.print("Enter price to search: ");
        double search = sc.nextDouble();

        int result = Arrays.binarySearch(prices, 0, size, search);

        if (result >= 0) {
            System.out.println("Price found at index: " + result);
        } else {
            System.out.println("Price not found.");
        }

        sc.close();
    }
}






class DoublyLinkedListPalindrome {
    private class Node {
        int data;
        Node prev, next;

        Node(int value) {
            data = value;
            prev = next = null;
        }
    }
    private Node head, tail;

    public void insert(int value) {
        Node newNode = new Node(value);
        if (head == null) { // list empty
            head = tail = newNode;
        } else {
            tail.next = newNode;
            newNode.prev = tail;
            tail = newNode;
        }
    }
    public boolean isPalindrome() {
        if (head == null) return true;
        Node left = head;
        Node right = tail;

        while (left != null && right != null && left != right && right.next != left) {
            if (left.data != right.data) {
                return false;
            }
            left = left.next;
            right = right.prev;
        }
        return true;
    }
    public void display() {
        Node current = head;
        System.out.print("DLL: ");
        while (current != null) {
            System.out.print(current.data + " ");
            current = current.next;
        }
        System.out.println();
    }
    public static void main(String[] args) {
        DoublyLinkedList dll = new DoublyLinkedList();
        dll.insert(1);
        dll.insert(2);
        dll.insert(3);
        dll.insert(2);
        dll.insert(1);
        dll.display();
  if (dll.isPalindrome()) {
            System.out.println("The Doubly Linked List is a Palindrome.");
        } else {
            System.out.println("The Doubly Linked List is NOT a Palindrome.");
        }
    }
}





public class Stack {
    int[] stack = new int[10];
    int top = -1;

    void push(int x) {
        if (top == 9) System.out.println("Stack is full");
        else stack[++top] = x;
    }

    int pop() {
        if (isEmpty()) {
            System.out.println("Stack is empty");
            return -1;
        }
        return stack[top--];
    }

    boolean isFull() { return top == 9; }
    boolean isEmpty() { return top == -1; }

    void printStack() {
        for (int i = top; i >= 0; i--) System.out.println(stack[i]);
    }

    void sortStack() {
        for (int i = 0; i <= top; i++) {
            for (int j = 0; j < top - i; j++) {
                if (stack[j] > stack[j + 1]) {
                    int temp = stack[j];
                    stack[j] = stack[j + 1];
                    stack[j + 1] = temp;
                }
            }
        }
    }

    void checkPalindrome(int num) {
        int original = num, reverse = 0;
        while (num > 0) {
            int digit = num % 10;
            reverse = reverse * 10 + digit;
            num /= 10;
        }
        if (original == reverse) System.out.println("Palindrome");
        else System.out.println("Not Palindrome");
    }

    public static void main(String[] args) {
        Stack s = new Stack();
        s.push(1); s.push(3); s.push(5); s.push(2);
        s.push(5); s.push(3); s.push(1);
        System.out.println("Before sorting:");
        s.printStack();
        s.sortStack();
        System.out.println("After sorting:");
        s.printStack();
        s.checkPalindrome(1331);
    }
}






public class LinkedList {
    class Node {
        String data;
        Node next;
        Node(String value) { data = value; }
    }
    Node head = null;
    void insert(String value) {
        Node newNode = new Node(value);
        if (head == null) head = newNode;
        else {
            Node temp = head;
            while (temp.next != null) temp = temp.next;
            temp.next = newNode;
        }
    }
    void insertFirst(String value) {
        Node newNode = new Node(value);
        newNode.next = head;
        head = newNode;
    }
    void delete(String value) {
        if (head == null) {
            System.out.println("List is empty");
            return;
        }
        if (head.data.equals(value)) {
            head = head.next;
            return;
        }
        Node temp = head;
        while (temp.next != null) {
            if (temp.next.data.equals(value)) {
                temp.next = temp.next.next;
                return;
            }
            temp = temp.next;
        }
        System.out.println("Element not found");
    }
    void search(String value) {
        Node temp = head;
        while (temp != null) {
            if (temp.data.equals(value)) {
                System.out.println("Found");
                return;
            }
            temp = temp.next;
        }
        System.out.println("Not Found");
    }
    void display() {
        Node temp = head;
        while (temp != null) {
            System.out.print(temp.data + " ");
            temp = temp.next;
        }
        System.out.println();
    }
    boolean isPalindrome(String word) {
        LinkedList reversed = new LinkedList();
        for (int i = 0; i < word.length(); i++)
            reversed.insertFirst(word.substring(i, i + 1));

        Node temp = reversed.head;
        for (int i = 0; i < word.length(); i++) {
            if (!word.substring(i, i + 1).equals(temp.data)) return false;
            temp = temp.next;
        }
        return true;
    }
    public static void main(String[] args) {
        LinkedList list = new LinkedList();
        list.insert("1"); list.insert("2"); list.insert("3");
        list.insert("4"); list.insert("5");

        System.out.println("Linked list:");
        list.display();
        list.delete("3");
        System.out.println("After deletion:");
        list.display();
        System.out.print("Search for 5: ");
        list.search("5");

        if (list.isPalindrome("radar"))
            System.out.println("radar is a palindrome");
        else
            System.out.println("radar is not a palindrome");
    }
}
